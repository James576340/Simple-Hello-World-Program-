#include <stdio.h>

int main() {
    FILE *fptr;
    char title[100];

    fptr = fopen("borrowed_books.txt", "a"); 
    if (fptr== NULL) {
        printf("Error opening file!\n");
        return 1;
    }

    printf("Enter book title: ");
    fgets(title, sizeof(title), stdin);  

    fputs(title, fptr);  
    fclose(fptr);

    printf("Book title successfully stored in borrowed_books.txt\n");
    return 0;
}
